//Selector methods are listed below:
  
  1. document.getElementByID('Id_name_here')
    -> document.getElementByID('Id_name_here').id    // select the id
    -> document.getElementByID('Id_name_here').className    // select the class
    -> document.getElementByID('Id_name_here').getAttribute('id') or any attributes   // select the attributes
    -> document.getElementByID('Id_name_here').setAttribute('class', 'class_name_to_set')[try to prevent previous value]
    -> document.getElementByID('Id_name_here').innerHTML = <h1>Modify the Heading contents</h1>

  2. document.getElementByClassName('Id_name_here')
  3. textContent  // getting all the actual available text [jo bhi actual mei present hai sb dikhega]
  4. innerText   // getting Only actual visible text [don't get the hidden text, hided using CSS]
  5. innerHTML   // getting all available text including html, if any html present inside. [span tag inside h1 tag]
